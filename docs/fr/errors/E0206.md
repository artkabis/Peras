# E0206 : `&` et les arguments `inout`

_Version anglaise, qui fait foi : [../../en/errors/E0206.md](../../en/errors/E0206.md)._

Un argument passé à un paramètre `inout` s’écrit avec `&`, et `&` ne s’écrit que pour ces arguments. Il montre à l’appel que la fonction peut modifier la variable.

## Exemple

```peras
fn clear(xs: inout Array<i32>): void {
  while (xs.length > 0) {
    xs.pop();
  }
}

export fn empty(): i32 {
  let xs: Array<i32> = [1, 2];
  clear(xs);
  return xs.length;
}
```

## Correction

```peras
fn clear(xs: inout Array<i32>): void {
  while (xs.length > 0) {
    xs.pop();
  }
}

export fn empty(): i32 {
  let xs: Array<i32> = [1, 2];
  clear(&xs);
  return xs.length;
}
```
