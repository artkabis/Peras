# E0222: Match does not cover every value

_French translation: [../../fr/errors/E0222.md](../../fr/errors/E0222.md)._

A `match` must handle every value its scrutinee can have: each variant of an enumeration or an option, `true` and `false`, and for numbers or strings a final `_`. When a variant is added to an enumeration, every `match` that forgets it is reported: that is the point of exhaustiveness.

## Example

```peras
enum Shape {
  Circle(f64),
  Rect(f64, f64),
  Empty,
}

export fn area(s: Shape): f64 {
  return match (s) {
    Circle(r) => 3.14159 * r * r,
    Rect(w, h) => w * h,
  };
}
```

## Fix

```peras
enum Shape {
  Circle(f64),
  Rect(f64, f64),
  Empty,
}

export fn area(s: Shape): f64 {
  return match (s) {
    Circle(r) => 3.14159 * r * r,
    Rect(w, h) => w * h,
    Empty => 0.0,
  };
}
```
