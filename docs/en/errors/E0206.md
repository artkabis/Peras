# E0206: `&` and `inout` arguments

_French translation: [../../fr/errors/E0206.md](../../fr/errors/E0206.md)._

An argument passed to an `inout` parameter is written with `&`, and `&` is written for those arguments only. It shows at the call that the function may change the variable.

## Example

```peras
fn clear(xs: inout Array<i32>): void {
  while (xs.length > 0) {
    xs.pop();
  }
}

export fn empty(): i32 {
  let xs: Array<i32> = [1, 2];
  clear(xs);
  return xs.length;
}
```

## Fix

```peras
fn clear(xs: inout Array<i32>): void {
  while (xs.length > 0) {
    xs.pop();
  }
}

export fn empty(): i32 {
  let xs: Array<i32> = [1, 2];
  clear(&xs);
  return xs.length;
}
```
