# E0207: Aliasing with an `inout` argument

_French translation: [../../fr/errors/E0207.md](../../fr/errors/E0207.md)._

Two arguments of a call reach the same variable, and one of them is `inout`. An `inout` parameter changes its variable in place: another argument reaching it would see it change during the call.

## Example

```peras
fn append(target: inout Array<i32>, source: Array<i32>): void {
  for (const x of source) {
    target.push(x);
  }
}

export fn doubled(): i32 {
  let xs: Array<i32> = [1, 2];
  append(&xs, xs);
  return xs.length;
}
```

## Fix

```peras
fn append(target: inout Array<i32>, source: Array<i32>): void {
  for (const x of source) {
    target.push(x);
  }
}

export fn doubled(): i32 {
  let xs: Array<i32> = [1, 2];
  append(&xs, xs.copy());
  return xs.length;
}
```
