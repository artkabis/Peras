# E0227: A type argument that does not meet a constraint

_French translation: [../../fr/errors/E0227.md](../../fr/errors/E0227.md)._

A generic function works for every type that meets the constraints of its parameters, and only for those: its body relies on them. Here `larger` compares its arguments, so `T` must be `Ord`, and strings are not ordered. Pass values whose type meets the constraint.

## Example

```peras
fn larger<T: Ord>(a: T, b: T): T {
  if (a > b) {
    return a;
  }
  return b;
}

export fn longest(a: String, b: String): String {
  return larger(a, b);
}
```

## Fix

```peras
fn larger<T: Ord>(a: T, b: T): T {
  if (a > b) {
    return a;
  }
  return b;
}

export fn longest(a: String, b: String): i32 {
  return larger(a.length, b.length);
}
```
